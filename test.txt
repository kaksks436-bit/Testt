local function consent(body)
    local t={}
    for tag in body:gmatch('<input type="hidden" name=".-" value=".-">') do
        local k,v=tag:match('<input type="hidden" name="(.-)" value="(.-)">');
        t[k]=v
    end
    gv=t.v or "";
    if writefile then writefile("googlev.txt",gv) end
end

local function got(url,method,body)
    method=method or "GET"
    local res=req({Url=url,Method=method,Headers={cookie="CONSENT=YES+"..(gv or "")},Body=body})
    local b=res.Body or res.body or "";
    if type(b)~="string" then b=tostring(b) end
    if b:match("https://consent.google.com/s") or b:match("consent") then
        consent(b)
        res=req({Url=url,Method=method,Headers={cookie="CONSENT=YES+"..(gv or "")},Body=body})
    end
    return res
end

local function iso(s)
    if not s then return end

local function q(data)
    local s=""
    for k,v in pairs(data) do
        if type(v)=="table" then for _,vv in pairs(v) do s..="&"..Http:UrlEncode(k).."="..Http:UrlEncode(vv) end
        else s..="&"..Http:UrlEncode(k).."="..Http:UrlEncode(v) end
    end
    return s:sub(2)
end

local function initializeUI()
    if isInitialized then return end
    screenGui = Instance.new("ScreenGui")
    screenGui.Name = "EnhancedNotifUI"
    screenGui.Parent = playerGui
    screenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    screenGui.DisplayOrder = 999
    screenGui.ResetOnSpawn = false
    isInitialized = true
end

local function updateNotificationPositions()
    if not screenGui then return end
    local currentY = -CONFIG.Padding
    local itemsToRemove = {}
    for i = 1, #notificationList do
        local notifFrame = notificationList[i]
        if not notifFrame or not notifFrame.Parent then
            table.insert(itemsToRemove, i)
            continue
        end
        local targetPos = UDim2.new(1, -CONFIG.Padding, 1, currentY)
        notifFrame.Position = targetPos
        currentY = currentY - (notifFrame.AbsoluteSize.Y + CONFIG.Padding)
    end
    for i = #itemsToRemove, 1, -1 do
        table.remove(notificationList, itemsToRemove[i])
    end
end

local function handleConvert()
    local text = TextBox.Text
    if text == "" then return end
    

    if auto_translate_on then
        local translated_text = translate(text, tr_lang, "auto")
        if translated_text and translated_text ~= "" then
            text = translated_text
            NotificationLibrary.Info(" Message translated: " .. text, "Translation")
        end
    end
    

    text = text:gsub("۔߻ ", ""):gsub("۔﹙߻﹚۔", "")
    local leadPunc = text:match("^[%p%s]+")
    if leadPunc then
        text = text:sub(#leadPunc + 1) .. leadPunc
    end
   
    local cleanedText = ""
    local i = 1
    while i <= #text do
        local found = false
        for _, bypass in ipairs(sortedReverseKeys) do
            if text:sub(i, i + #bypass - 1) == bypass then
                cleanedText = cleanedText .. ReverseMap[bypass]
                i = i + #bypass
                found = true
                break
            end
        end
        if not found then
            cleanedText = cleanedText .. text:sub(i, i)
            i = i + 1
        end
    end
    cleanedText = string.lower(cleanedText)
    local words = string.split(cleanedText, " ")
    local finalResult = {}
    for _, word in ipairs(words) do
        local prefix, core, suffix = word:match("^([^%w]*)(.-)([^%w]*)$")
        if core and WordMap[core] then
            table.insert(finalResult, (prefix or "") .. WordMap[core] .. (suffix or ""))
        else
            local bypassedWord = ""
            for j = 1, #word do
                local c = word:sub(j, j)
                bypassedWord = bypassedWord .. (CharMap[c] or c)
            end
            table.insert(finalResult, bypassedWord)
        end
    end
    local out = table.concat(finalResult, " ")
    if symbolOn then
        local endPunc = out:match("[%p%s]+$") or ""
        if endPunc ~= "" then
            out = endPunc .. out:sub(1, #out - #endPunc)
        end
        out = "۔߻ " .. out .. "۔﹙߻﹚۔"
    end
    
    TextBox.Text = out
    NotificationLibrary.Info("Test", "successfully")
end
